    \,\parbox{4pt}{%
        \begin{tikzpicture}%
            \draw[line width=.7pt] (.2ex, .8ex) -- (.8ex, .2ex);%
            \node[scale=0.0] at (0,0) {$\setminus$};
        \end{tikzpicture}%
    }\,%
}

                (1ex, .5ex)
                (1.33ex, .07ex)
                (2.05ex, .5ex)
                (2.6ex, .5ex)
            };%

        \end{tikzpicture}%
