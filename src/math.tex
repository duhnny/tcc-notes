% aesthetics
\def\smallsetminus{
    \,\parbox{4pt}{%
        \begin{tikzpicture}%
            \draw[line width=.7pt] (.2ex, .8ex) -- (.8ex, .2ex);%
            \node[scale=0.0] at (0,0) {$\setminus$};
        \end{tikzpicture}%
    }\,%
}
\def\encodes{
    \,\parbox{13pt}{%
        \begin{tikzpicture}[scale=1.1]%
            \draw[line width=.55pt] plot [smooth, tension=1] coordinates {
                (.2ex, .5ex)
                (1.5ex, .5ex)
                (1.3ex, .9ex)
                (1ex, .5ex)
                (1.33ex, .07ex)
                (2.05ex, .5ex)
                (2.6ex, .5ex)
            };%
            \draw[line width=.55pt] (2.2ex, .9ex)  -- (2.62ex, .48ex);%
            \draw[line width=.55pt] (2.2ex, .14ex) -- (2.62ex, .56ex);%
        \end{tikzpicture}%
    }\,%
}
\def\liftingProperty{
    \,\parbox{9pt}{%
        \begin{tikzpicture}[scale=1.1]%
            \draw (0ex,0) rectangle (1.5ex,1.5ex);%
            \draw (0ex,0) -- (1.5ex,1.5ex);%
        \end{tikzpicture}%
    }\,%
}
