    Bom, o primeiro passo para isso é mostrar que, de fato, está é uma categoria.

    \begin{theorem}\mylabel{graph_is_a_category}
        $(\Graph, \Enc)$ é uma categoria
    \end{theorem}
    \begin{proof}
        (Associatividade) Sejam $A, B, C \in \Graph$ e seja $f \in \Hom{A}{B}$ e $g \in \Hom{B}{C}$,
        então $g \circ f \in \Hom{A}{C}$.

        De fato, seja $c_1 c_2$ uma aresta em $C$. Então, pela definição de encoding, existem
        $b_1 \in g^{-1}(\{c_1\})$ e $b_2 \in g^{-1}(\{c_2\})$ tais que $b_1 b_2 \in \Edg{C}$.

        Repetindo o argumento para $A$, encontramos dois vértices $a_1, a_2 \in \Ver{A}$ tais que
        $f(a_1) = b_1$, $f(a_2) = b_2$ e $a_1 a_2 \in \Edg{A}$. Logo, $a_1 a_2$ é uma aresta em
        $A$ tal que

        \[(g \circ f) (a_1) = g(f(a_1)) = g(b_1) = c_1\]

        e analogamente para $C$.
    \end{proof}

    Munidos do conhecimento de que o nosso objeto de estudo é uma categoria, podemos também definir
    um produto, que nos será muito útil mais afrente.

    \begin{definition}\mylabel{product}
        Sejam $A, B \in \Graph$ grafos. Definimos então o grafo produto $A \times B$ como sendo o
        grafo com vértices $\Ver{A} \times \Ver{B}$ e em que, se $(a, b)$ e $(a', b')$ são
        dois vértices de $A \times B$, há uma aresta entre os dois se, e somente se,
        $a = a' \wedge b b' \in \Edg{B}$ ou $a a' \in \Edg{A} \wedge b = b'$.

        De forma mais geral, sejam $A_1, A_2, \cdots A_n \in \Graph$, então o grafo produto
        $\prod_{i} A_i$ é o grafo no conjunto de vértices $\prod_i A_i$ definido da seguinte forma:

        \begin{equation*}
            (a_1, a_2, \cdots a_n) (a_1', a_2', \cdots a_n') \in \Edg{\prod_i A_i} \iff
            \exists i \in \{1, \cdots n\}
            \left( a_i a_i' \in \Edg{A_i} \wedge a_j = a_j', \forall j \neq i \right)
        \end{equation*}
        ou seja, se os vértices diferem em no máximo uma coordenada, e, nessa coordenada, há uma
        aresta entre os dois vértices.
    \end{definition}

    Basta mostrar que esse produto realmente é um produto de categorias.

    Lembre que, em uma categoria $(C, Hom)$, um produto $X \defeq \prod_i X_i$ é definido como um
    elemento de $C$, munido de morfismos $\pi_i: X \to X_i$ tais que, dada uma família de
    morfismos $f_i: Y \to X_i$, há um único morfismo $f: Y \to X$ tal que o seguinte diagrama
    comuta:

    \[
        \begin{tikzcd}

\begin{lemma}\mylabel{prod_kayley}

    \begin{equation*}
        \Kayley{G}{S} \boxProduct \Kayley{H}{T}

\begin{proof}
    $(a, b) \to (a', b')$ é uma aresta de
    $\Kayley{G \oplus H}{(S \times \{e_H\}) \cup (\{e_G\} \times T)}$ se, e somente se, existe
    $c \in S \times \{e_H\} \cup \{e_G\} \times T$ tal que $(a',b') = (a, b) \cdot c$. Mas isso

    $(a, b) \to (a', b')$ é aresta de $\Kayley{G}{S} \boxProduct \Kayley{H}{T}$.

    Voltando o argumento, temos o outro lado do isomorfismo.
\end{proof}

    $f \colon G \to H$ um homomorfismo sobrejetor de grupoides tal que $f(S) \supseteq T$. Então

        \Kayley{G}{S} \encodes \Kayley{H}{T}

\begin{proof}
    Dadas essas condições, construa, parcialmente, o seguinte homomorfismo $\bar{f}$ de grafos:

    \begin{equation*}
        \bar{f}(a) = f(a),\hspace{5pt} \forall a \in \Ver{\Kayley{G}{S}} = G
    \end{equation*}
    e, para toda aresta $a \to b$ de $\Kayley{G}{S}$, sabemos que existe $s \in S$ tal que
    $a \cdot s$. Nesse caso, se $f(s) \in T$, definimos

    \begin{equation*}
        f(a \to b) = f(a) \to f(b) = f(a \cdot s) = f(a) \cdot f(s)
    \end{equation*}
    que está em $\Edg{\Kayley{H}{T}}$ porque $f(s) \in T$. Caso contrário, não definimos $\bar{f}$
    sobre a aresta.

    Agora, para mostrar que $\bar{f}$ é encoding, tome $b \in G$ e uma aresta
    $a \to \bar{f}(b) = f(b)$. Então, sabemos que $f(b) = a \cdot t$ para algum $t \in T$. Mas
    $f(S) \supseteq T$. Logo, existe um $s \in S$ tal que $f(s) = t$.

    Tome, então, $b \cdot s^{-1}$. Temos, pois, 
    
    \begin{align*}
        f(b \cdot s^{-1}) &= f(b) \cdot s^{-1} \\
                          &= a \cdot s \cdot s^{-1} \\
                          &= a
    \end{align*}
    e $b \cdot s^{-1} \to b$ é mapeada para $a \to f(b)$ por $\bar{f}$.
\end{proof}

\begin{theorem}\mylabel{normal_enc_kayley}
    Seja $G$ um grupo, $N_i$ uma família de subgrupos normais de $G$ tais que $G = \prod_i N_i$ e
    $S \subseteq G$ um subconjunto de $G$ qualquer tal que para todo $s \in S$, existe $i$ tal que
    $s \in N_i$.

    \begin{equation*}
        \bigBoxProduct_i \Kayley{N_i}{*} \encodes \Kayley{G}{S}
    \end{equation*}
\end{theorem}

\begin{proof}
    Defina $f \colon \bigBoxProduct_i \Kayley{N_i}{*} \to \Kayley{G}{S}$ parcialmente levando cada
    vértice $(n_i)_{i \in I}$ no produto $\prod_i n_i \in G$. Além disso, para cada aresta, que é
    da forma

    \begin{equation*}
        (n_0, n_1, \cdots, n_i, \cdots, n_k) \to (n_0', n_1', \cdots, n_i', \cdots, n_k')
    \end{equation*}
    em que $n_j = n_j', \forall j \neq i$, se houver $s \in S$ tal que
    $n_0' \cdots n_k' = s n_0 \cdots n_k$, então a aresta acima é mapeada na aresta

    \begin{equation*}
        n_0 n_1 \cdots n_k \to s n_0 n_1 \cdots n_k = n_0' n_1' \cdots n_k'
    \end{equation*}
    e, se não houver, não é definida.

    Provaremos que $f$ assim dada é um encoding. Nesse caso, dado que $f$ é sobrejetora nos
    vértices porque $\prod_i N_i = G$, basta mostrar que, para todo ponto
    $n \in \bigBoxProduct_i N_i$ e para toda aresta $g \to f(n)$ de $\Kayley{G}{S}$, há um
    levantamento da aresta dado por $f$.

    De fato, podemos escrever a aresta acima como $g \to sg$, para algum $s \in S$, por construção
    de $\Kayley{G}{S}$. Ou seja, $f(n) = sg$ para algum $s \in S$. Sabemos, então, que, dado que
    $G = \prod_i N_i$, podemos escrever

    \begin{equation*}
        g = n_0 n_1 \cdots n_k
    \end{equation*}
    com cada $n_i \in N_i, \forall i$.

    Além disso, sabemos que existe um $i$ tal que $s \in N_i$. Fixe tal $i$.

    Temos, então, que

    \begin{align*}
        s n_0 n_1 \cdots n_i \cdots n_k &= ( n_0 n_1 \cdots n_{i-1} )
                                           ( n_0 n_1 \cdots n_{i-1} )^{-1}
                                           \cdot ( s n_0 n_1 \cdots n_i \cdots n_k ) \\
                                        &= ( n_0 n_1 \cdots n_{i-1} )
                                           \left( ( n_0 n_1 \cdots n_{i-1} )^{-1}
                                           s ( n_0 n_1 \cdots n_{i-1} ) \right)
                                           \cdot n_i \cdots n_k \\
    \end{align*}

    Como $N_i$ é normal, temos que

    \begin{equation*}
        \left( n_0 n_1 \cdots n_{i-1} \right)^{-1}
            \cdot s
            \cdot \left( n_0 n_1 \cdots n_{i-1} \right) \in N_i
    \end{equation*}
    \begin{equation*}
         \implies \left( n_0 n_1 \cdots n_{i-1} \right)^{-1}
                    \cdot s
                    \cdot \left( n_0 n_1 \cdots n_{i-1} \right)
                    \cdot n_i
                    =: n_i' \in N_i
    \end{equation*}

    Logo,

    \begin{equation*}
        s n_0 n_1 \cdots n_i \cdots n_k = n_0 n_1 \cdots n_i' \cdots n_k
    \end{equation*}
    ou seja, a aresta

    \begin{equation*}
        (n_0, n_1, \cdots, n_i, \cdots n_k) \to (n_0, n_1, \cdots, n_i', \cdots n_k)
    \end{equation*}
    é mapeada por $f$ para a aresta

    \begin{equation*}
        g \to s g = f(n)
    \end{equation*}
\end{proof}

\begin{lemma}\mylabel{li_finite_field}

\begin{theorem}\mylabel{prod_enc_prod}
    Sejam $A, B, C, D \in \Graph$ grafos tais que $A \encodes B$ e $C \encodes D$. Então

    \begin{equation*}
        A \boxProduct C \encodes B \boxProduct D
    \end{equation*}
\end{theorem}

\begin{proof}
    Sejam $f \colon A \to B$ e $g \colon C \to D$ encodings. Além disso, seja tome
    $(a_2, c_2) \in \Obj{A \boxProduct C}$.

    Seja, além disso, $(b_2, d_2) \defeq (f(a_2), g(c_2))$.

    Suponha, então que

    \begin{equation*}
        (b_1, d_1) \to (b_2, d_2)
    \end{equation*}
    seja umama aresta de $B \boxProduct D$. Então $b_1 = b_2$ ou $d_1 = d_2$. Escolha, sem perda de
    generalidade, $b_1 = b_2$. Então, pela construção de $B \boxProduct D$, existe uma aresta
    $d_1 \to d_2 = g(d_1)$ em $D$. Logo, pela definição de encoding, existe um $c_1 \in \Obj{C}$
    e uma aresta $c_1 \to c_2$ em $C$ levada em $d_1 \to d_2$ em $D$.

    Além disso, como $f$ é sobrejetora, temos que existe $a_1 \in f^{-1}(\{b_1\})$. Logo,

    \begin{equation*}
        (a_1, c_1) \to (a_1, c_2)
    \end{equation*}
    é uma aresta em $A \boxProduct C$ que é levada em $(b_1, d_1) \to (b_2, d_2)$.
\end{proof}

\begin{theorem}[Teorema 2.10]\mylabel{factor_mu_1}
    Seja $n = p_1^{r_1} \cdots p_k^{r_k}$ e suponha que $s_1, \cdots, s_k \in \N$ sejam tais que
    $s_i | r_i, \forall i$. Então, tomando $m = p_1^{s_1} \cdots p_k^{s_k}$, temos que

    \begin{equation}
        \mu_1(n) \leq
            \Big(\prod_{i = 1}^k \frac{p^{s_i}}{p^{s_i} - 1}\Big) \cdot \frac{\mu_1(m)}{m} \cdot n
    \end{equation}
\end{theorem}

\begin{proof}
    Seja $G \defeq \prod_i GF(p_i^{r_i})$. Repare que $\card{G} = n$. Logo,
    $\Kayley{G}{*} \cong K_n$. Mas, segundo o \myref[Lema]{li_finite_field}, para cada $i$ existe
    $S_i \subseteq GF(p_i^{r_i})$, um conjunto de vetores linearmente independentes sobre
    $GF(p_i^{s_i})$ que geram $GF(p_i^{r_i})$ pela ação de $GF(p_i^{s_i})$ em $S_i$.
    
    Sejam então

    \begin{equation}
        G' \defeq \prod_i GF(p_i^{s_i})
    \end{equation}
    e

    \begin{equation}
        S \defeq \prod_i S_i \subseteq G' 
    \end{equation}

    Então cada elemento de $G$ pode ser escrito como $s g'$, em que $s \in S$ e $g' \in G'$.

    Considere, agora, $G$ como um grupo aditivo e as órbitas $sG' \subseteq G$ como subgrupos.
    Então é claro que os subgrupos $sG'$ são normais e, além disso, temos que

    \begin{equation*}
        G \subseteq \sum_{s \in S} sG'
    \end{equation*}

    Logo, pelo \myref[Teorema]{normal_enc_kayley}, temos que

    \begin{equation*}
        \bigBoxProduct_{s \in S} \Kayley{s G'}{*} \encodes \Kayley{G}{*} \cong K_n
    \end{equation*}
    
    Mas sabemos também que cada $s G'$ contém $m$ elementos. Sendo assim, temos
    $\Kayley{s G'}{*} \cong K_{m}$ para todo $s \in S$. Mas isso significa que

    \begin{equation*}
        Q_{\mu(m)} \encodes K_m
    \end{equation*}
    por definição de $\mu$.

    Logo, pelo \myref[Teorema]{prod_enc_prod}, nós temos que

    \begin{equation*}
        \bigBoxProduct_{s \in S} Q_{\mu(m)} \cong Q_{\card{S} \mu(m)} \encodes
        \bigBoxProduct_{s \in S} K_m \cong \bigBoxProduct_{s \in S} \Kayley{s G'}{*} \cong K_n
    \end{equation*}
    donde, pela definição de $\mu$, tiramos imediatamente que

    \begin{align*}
        \mu(n) &\leq \card{S} \mu(m) \\
               &=    \left( \prod_i \frac{p^{r_i} - 1}{p^{s_i} - 1} \right) \mu(m) \\
               &\leq \left( \prod_i \frac{p^{r_i}}{p^{s_i} - 1} \right) \mu(m) \\
               &=    n \left( \prod_i \frac{1}{p^{s_i} - 1} \right) \mu(m) \\
               &=    \frac{n}{m} \left( \prod_i \frac{p^{s_i}}{p^{s_i} - 1} \right) \mu(m)
    \end{align*}

    Isto é,
    \begin{equation*}
        \mu(n) \leq \left( \prod_i \frac{p^{s_i}}{p^{s_i} - 1} \right) \frac{\mu(m)}{m} n
    \end{equation*}
\end{proof}

\begin{lemma}\mylabel{prod_construction}
    Seja $r = r_1 + r_2$ e $n = n_1 + n_2$. Então

    \begin{equation}
        \mu_r(n) \leq \mu_{r_1}(n_1) + \mu_{r_2}(n_2)
    \end{equation}
\end{lemma}

\begin{proof}
    Sejam $m_1 \defeq \mu_{r_1}(n_1)$ e $m_2 \defeq \mu_{r_2}(n_2)$ e sejam

    \begin{center}
        \begin{minipage}{.3\linewidth}
            \begin{equation*}
                f_1 \colon Q_{m_1}^{(r_1)} \to K_{n_1}
            \end{equation*}
        \end{minipage}
        \begin{minipage}{.3\linewidth}
            \begin{equation*}
                f_2 \colon Q_{m_2}^{(r_2)} \to K_{n_2}
            \end{equation*}
        \end{minipage}
    \end{center}
    encodings.

    Defina, então, a função $f \colon (Q_{m_1} \boxProduct Q_{m_2})^{(r)} \to K_{n_1} \times K_{n_2}$
    da seguinte forma: seja $(q_1', q_2')$ um vértice de $Q_{m_1} \boxProduct Q_{m_2}$. Definimos

    \begin{equation*}
        f((q_1', q_2')) \defeq (f_1(q_1'), f_2(q_2'))
    \end{equation*}
    que é sobrejetora nos vértices porque $f_1$ e $f_2$ o são.

    Além disso, para cada aresta de $(Q_{m_1} \boxProduct Q_{m_2})^{(r)}$, escolha a (única) aresta
    de $K_{n_1} \times K_{n_2}$ que liga os pontos aos quais são mapeados o ponto inicial e final
    da aresta em questão.

    Isso define um homomorfismo de grafos. Provaremos que $f$ é encoding.

    Com efeito, se $f((q_1', q_2')) = (k_1', k_2')$ e existe uma aresta
    $(k_1, k_2) \to (k_1', k_2')$, então existe uma aresta $k_1 \to k_1'$ em $K_{n_1}$ e uma aresta
    $k_2 \to k_2'$ em $K_{n_2}$. Mas, dado que $f_1$ é encoding, significa que existe um vértice
    $q_1 \in \Obj{Q_{m_1}}^{(r_1)}$ e uma aresta $q_1 \to q_1'$ no mesmo grafo que mapeia para
    $k_1 \to k_1'$. Ou, equivalentemente, existe um caminho $q_1 \leadsto q_1'$ de comprimento
    menor ou igual a $r_1$ em $Q_{m_1}$ que é mapeado para $k_1 \to k_1'$.

    Escreva

    \begin{equation*}
        q_1 \to s_1 \to s_2 \to \cdots \to s_j \to q_1'
    \end{equation*}
    tal caminho em $Q_{m_1}$.

    Podemos traçar um caminho análogo em $Q_{m_1} \boxProduct Q_{m_2}$ mantendo a segunda
    coordenada e seguindo o caminho na primeira:

    \begin{equation*}
        (q_1, q_2') \to (s_1, q_2') \to (s_2, q_2') \to \cdots \to (s_j, q_2') \to (q_1', q_2')
    \end{equation*}
    que existe pela construção do produto caixa. Por construção, esse caminho tem comprimento menor
    ou igual a $r_1$.

    Seguindo, agora, os mesmos passos na segunda coordenada, podemos traçar o caminho

    \begin{equation*}
        (q_1, q_2) \to (q_1, t_1) \to \cdots \to (q_1, t_k) \to (q_1, q_2')
    \end{equation*}
    de comprimento menor ou igual a $r_2$.

    Concatenando os caminhos, temos um caminho $(q_1, q_2) \leadsto (q_1', q_2')$ de comprimento
    menor ou igual a $r_1 + r_2 = r$, ou, equivalentemente, uma aresta em
    $(Q_{m_1} \boxProduct Q_{m_2})^{(r)}$ que mapeia para $(k_1, k_2) \to (k_1', k_2')$.

    Logo, $f$ é encoding. Compondo $f$ com os isomorfismos

    \begin{center}
        \begin{minipage}{.3\linewidth}
            \begin{equation*}
                \left( Q_{m_1} \boxProduct Q_{m_2} \right)^{(r)} \cong Q_{m}^{(r)}
            \end{equation*}
        \end{minipage}
        \begin{minipage}{.3\linewidth}
            \begin{equation*}
                K_{n_1} \times K_{n_2} \cong K_n
            \end{equation*}
        \end{minipage}
    \end{center}
    garante o resultado.
\end{proof}

\begin{lemma}[Um limite exponencial]\mylabel{r_exp_limit}
    Sejam $n, r \geq 1$ inteiros. Então

    \begin{equation}
        \mu_r(n) \leq \alpha n^{\frac{1}{r}}
    \end{equation}
    em que

    \begin{equation*}
        \alpha \defeq 2^{1 + \frac{1}{r}} \cdot \frac{r}{e \cdot \ln 2}
    \end{equation*}
\end{lemma}

\begin{proof}
    Tome $n = 2^k$. Seja $k_1$ o resto da divisão de $k$ por $r$. Então $k = rs + k_2$, com
    $0 \leq k_2 \leq r - 1$. Escreva $k_1 \defeq r - k_2$. Então

    \begin{align*}
        k &= rs + k_2 \\
          &= (k_1 + k_2)s + k_2 \\
          &= k_1 s + k_2 s + k_2 \\
          &= k_1 s + k_2 (s + 1)
    \end{align*}

    Usando então o \myref[lema]{prod_construction}, podemos escrever:

    \begin{align*}
        \mu_r(n) &=    \mu_r(2^k) \\
                 &=    \mu_{k_1 + k_2}\left( 2^{k_1 s + k_2 (s + 1)} \right) \\
                 &\leq \mu_{k_1}(2^{k_1 s}) + \mu_{k_2}(2^{k_2 (s+1)}) \\
                 &\leq k_1 \mu_1(2^s) + k_2 \mu_1(2^{s+1})
    \end{align*}

    Agora, sabemos que $\mu_1(2^l) = 2^l - 1$. Logo,

    \begin{align*}
        \mu_r(n) &\leq k_1 \mu_1(2^s) + k_2 \mu_1(2^{s+1}) \\
                 &\leq k_1 2^s + k_2 2^{s+1} \\
                 &\leq (k_1 + 2k_2) 2^s \\
                 &=    (k_1 + 2k_2) (2^{rs})^{\frac{1}{r}} \\
                 &=    (k_1 + 2k_2) \left( \frac{2^{rs + k_2}}{2^{k_2}} \right)^{\frac{1}{r}} \\
                 &=    (k_1 + 2k_2) \left( \frac{2^k}{2^{k_2}} \right)^{\frac{1}{r}} \\
                 &=    (k_1 + 2k_2) \left( \frac{n}{2^{k_2}} \right)^{\frac{1}{r}} \\
                 &=    \left( \frac{k_1 + 2k_2}{2^{\frac{k_2}{r}}} \right) n^{\frac{1}{r}}
    \end{align*}

    Olhemos para o primeiro termo nessa desigualdade, como uma função de $k_2$. Como $k_1 + k_2 = r$
    fixado, teremos a seguinte função:

    \begin{equation*}
        \lambda(k_2) = \frac{r + k_2}{2^{\frac{k_2}{r}}}
    \end{equation*}

    Derivando essa função em $k_2$, temos:

    \begin{align*}
        \frac{\mathrm{d} \lambda}{\mathrm{d} k_2}(k_2) &=
            2^{-\frac{k_2}{r}} - \frac{\ln 2}{r} \cdot \frac{r + k_2}{2^{\frac{k_2}{r}}} \\
        &=  2^{-\frac{k_2}{r}} \left( 1 - \frac{\ln 2 (r + k_2)}{r} \right)
    \end{align*}
    que é igual a zero se, e somente se:

    \begin{align*}
        k_2 &=      -r + \frac{r}{\ln 2} \\
            &=      r \left( \frac{1}{\ln 2} - 1 \right) \\
            &\qefed k_0
    \end{align*}

    Além disso, se $0 \leq k_2 \leq k_0$, a derivada de $\lambda$ é positiva, e, se
    $k_0 \leq k_2 \leq r-1$, a derivada de $\lambda$ é negativa. Logo, $k_0$ é um ponto máximo de
    $\lambda$.

    Agora, mesmo que o fator acima na inequação com $\mu_r(n)$ seja somente válido para valores
    inteiros de $k_2$, é claro que será menor ou igual ao máximo da função real $\lambda$ no
    intervalo em questão. Logo, podemos dizer que

    \begin{align*}
        \mu_r(n) &\leq \left( \frac{r + k_0}{2^{\frac{k_0}{r}}} \right) n^{\frac{1}{r}} \\
                 &\leq \left( \frac{r - r + \frac{r}{\ln 2}}{2^{\frac{1}{\ln 2} - 1}} \right) n^{\frac{1}{r}} \\
                 &=    2^{1 - \frac{1}{\ln 2}} \cdot \frac{r}{\ln 2} \cdot n^{\frac{1}{r}} \\
                 &=    2^{1 - \frac{\log_2 e}{\log_2 2}} \cdot \frac{r}{\ln 2} \cdot n^{\frac{1}{r}} \\
                 &=    \frac{2r}{e \cdot \ln 2} \cdot n^{\frac{1}{r}}
    \end{align*}

    Mas isso vale para $n = 2^k$. Agora, para qualquer $n \in \N$ existe um $k \in \N$ tal que
    $n \leq 2^k \leq 2n$ \TODO{adicionar prova disso?}. Mas isso significa que:

    \begin{align*}
        \mu_r(n) &\leq \mu_r(2^k) \\
                 &\leq \frac{2r}{e \cdot \ln 2} \cdot (2^k)^{\frac{1}{r}} \\
                 &\leq \frac{2r}{e \cdot \ln 2} \cdot (2n)^{\frac{1}{r}} \\
                 &=    2^{1 + \frac{1}{r}} \cdot \frac{r}{e \cdot \ln 2} \cdot n^{\frac{1}{r}}
    \end{align*}
    que nos dá o resultado.
\end{proof}

\begin{theorem}
    Para todos $\epsilon > 0$ e $r \in \N$, $r \geq 1$, existe um número $N \in \N$ tal que

    \begin{equation*}
        \mu_r(n) \leq (1 + \epsilon) r n^{\frac{1}{r}}
    \end{equation*}
\end{theorem}

\begin{proof}
    Fixe $\epsilon > 0$. Considere a diferença $\Delta(n, r) \defeq (n+1)^r - n^r$ entre a
    $n$-ésima $r$ potência e a sua sucessora. Pelo Teorema do Valor Médio aplicado à função
    $f(x) = x^r$, sabemos que a diferença

    \begin{align*}
        \Delta(n, r) &= (n+1)^r - n^r \\
                     &= \frac{(n+1)^r - n^r}{1} \\
                     &= f'(x_0)
    \end{align*}
    em que $x_0$ é algum ponto entre $n$ e $n+1$.

    Mas

    \begin{align*}
        f'(x_0) &=    r x^{r-1} \\
                &\leq r (n+1)^{r-1} \\
                &=    r (n^{r-1} + (r-1)n^{r-2} + \cdots + 1) \\
                &=    r n^{r-1} + r((r-1)n^{r-2} + \cdots + 1) \\
                &\leq r n^{r-1} + n^{r-1} \\
                &=    (r + 1) n^{r-1}
    \end{align*}
    para $n$ suficientemente grande.

    Seja, então, $N_1 \in \N$ tal que, para todo $n \geq N_1$, o que está acima valha. Então

    \begin{equation*}
        \Delta(n, r) \leq (r + 1) n^{r-1}, \hspace{10pt} \forall n \geq N_1
    \end{equation*}

    Mas isso implica que

    \begin{align*}
        \frac{\Delta(n, r)}{n^r} &\leq \frac{(r + 1) n^{r-1}}{n^r} \\
                                 &=    \frac{(r + 1)}{n} \\
                                 &<    \frac{\epsilon}{3}
    \end{align*}
    para $n$ suficientemente grande. Seja $N_2 \in \N$, $N_2 \geq N_1$ tal que a desigualdade acima
    com $\epsilon$ valha. Então, reescrevendo a desigualdade acima, temos:

    \begin{alignat*}{3}
        0   &< \quad \frac{\Delta(n, r)}{n^r} &&< \frac{\epsilon}{3} \\
        0   &< \frac{(n+1)^r - n^r}{n^r}      &&< \frac{\epsilon}{3} \\
        0   &< (n+1)^r - n^r                  &&< \frac{\epsilon}{3} n^r \\
        n^r &< \quad (n+1)^r                  &&< \left( 1 + \frac{\epsilon}{3} \right) n^r
    \end{alignat*}
    ou seja, para um $n$ grande o suficiente, existe uma $r$-potência entre $n^r$ e
    $(1 + \epsilon/3)n^r$.

    Tome $N_3 = N_2^r \geq N_2$. Agora tome um $n$ qualquer, $n \geq N_3$, e seja $m = k^r$ a
    última $r$-potência menor ou igual a $n$. Então $k \geq N_2$, porque $n \geq N_3 = N_2^r$.
    Sabemos, pelo exposto acima, que

    \begin{align*}
        k^r < (k+1)^r &<    \left( 1 + \frac{\epsilon}{3} \right) k^r \\
                      &\leq \left( 1 + \frac{\epsilon}{3} \right) n
    \end{align*}
    isto é, $(k+1)^r < (1 + \epsilon/3)n$. Mas, por hipótese, $k^r$ é a última $r$-potência menor
    ou igual a $n$. Logo, para todo $n \geq N_3$, existe uma $r$-potência entre $n$ e
    $(1 + \epsilon/3) n$.

    Agora, escolha, pelo \myref[Teorema]{lim_assintotico_mu_1} $N_4 \geq N_3$ tal que para todo
    $n \geq N_4^{1/r}$, tenhamos

    \begin{equation*}
        \mu_1(n) \leq \left( 1 + \frac{\epsilon}{3} \right) n
    \end{equation*}

    Por fim, se $n \geq N_4$, então existe uma $r$-potência $k^r$ entre $n$ e $(1 + \epsilon/3)n$.
    Então sabemos que

    \begin{center}
        \begin{minipage}{.3\linewidth}
            \begin{equation*}
                \mu_1(k) \leq \left( 1 + \frac{\epsilon}{3} \right) k
            \end{equation*}
        \end{minipage}
        \begin{minipage}{.3\linewidth}
            \begin{align*}
                         & k^r < \left( 1 + \frac{\epsilon}{3} \right) n \\
                \implies & k < \left( 1 + \frac{\epsilon}{3} \right)^{1/r} n^{1/r} \\
                \implies & k < \left( 1 + \frac{\epsilon}{3} \right) n^{1/r}
            \end{align*}
        \end{minipage}
    \end{center}

    Assim sendo,

    \begin{align*}
        \mu_r(n) &\leq \mu_r(k^r) \\
                 &\leq r \mu_1(k) \\
                 &\leq r \left( 1 + \frac{\epsilon}{3} \right) k \\
                 &<    r \left( 1 + \frac{\epsilon}{3} \right)^2 n^{1/r} \\
                 &\leq r \left( 1 + \epsilon \right) n^{1/r} \\
    \end{align*}

    Escolhendo $N = N_4$, temos, portanto, o resultado.
\end{proof}

Nesse contexto, a identidade faz o elemento neutro da operação de composição, e as composições de

Sabendo disso, olhemos de novo para a definição de encoding:

\begin{definition}\mylabel{encoding}
    Sejam $A, B \in \Graph$ grafos e $f \colon A \to B$ uma flecha, isto é, um homomorfismo entre
    eles. Então dizemos que $f$ é um \textbf{encoding} se, e somente se, $f$ é um epimorfismo e,
    para todo $b \in \Ver{A}$, e toda aresta $a \to_\alpha f(b)$ em $B$, existe um levantamento
    $a' \to_{\alpha'} b$ de $\alpha$, isto é, $\alpha' \in \Edg{A}$ e $f(\alpha') = \alpha$.
\end{definition}

Tomemos um momento para apreciar essa definição.

A princípio, parece que esta definição trata da mesma coisa duas vezes: a primeira dizendo que $f$
é um epimorfismo. No contexto de um topos, como a categoria que estamos lidando, isto é, $\Graph$,
dizer que temos um epimorfismo é equivalente a dizer que existe uma sobrejeção sobre os conjuntos
correspondentes -- no caso, $\Ver{A} \to \Ver{B}$ e $\Edg{A} \to \Edg{B}$ --, que respeita as
devidas relações.

Ora, se a segunda condição diz que há um levantamento para qualquer aresta do codomínio, isso não é
o mesmo que reafirmar o caráter de epimorfismo de $f$? Não, e justamente porque a segunda condição
não diz isso. A segunda condição da definição acima nos diz que dado um vértice $v$ de $A$ e uma
aresta terminando na imagem por $f$ de $v$, existe um levantamento dessa aresta \textit{que
termina em $v$}, e isso faz toda a diferença. Observe o exemplo a seguir:

    \end{enumerate}

\begin{definition}\mylabel{functor_fibration}
    Seja $F \colon A \to B$ um funtor entre duas categorias $A, B$. Diremos, então, que $F$ é uma
    \textbf{fibração} se, e somente se, para todo $y \in A$ e toda seta $f \colon x \to F(y)$ em
    $B$, existe uma seta $f' \colon x' \to y$ em $A$ tal que $F f' = f$.
\end{definition}

\begin{lemma}\mylabel{conservative_reflects_id}
    Sejam $A, B \in \Graph$ e $F \colon \Free(A) \to \Free(B)$ um funtor. Então, se $F$ é
    conservativo, então $F$ reflete identidades, isto é, se $f \colon a \to b$ é um morfismo em
    $\Free(A)$ tal que $F(f)$ é uma identidade em $\Free(B)$, então $a = b$ e $f = \id{a}$.
\end{lemma}

\begin{proof}
    Se $F(f)$ é uma identidade, então, em particular, é um isomorfismo. Portanto, dado que $F$ é
    conservativo, então $f$ também é isomorfismo. Mas, como os únicos isomorfismos de uma categoria
    livre são as identidades, então $f$ é identidade.
\end{proof}

\begin{theorem}
    Sejam $A, B \in \Graph$, e $f \colon A \to B$ um encoding de grafos. Então

    \begin{equation*}
        \Free f \colon \Free(A) \to \Free(B)
    \end{equation*}
    é uma fibração conservativa.

    Por outro lado, se $F \colon \Free(C) \to \Free(B)$ é uma fibração conservativa, existe
    um subgrafo $A \hookrightarrow C$ e um $f \colon A \to B$ encoding tal que o seguinte diagrama
    comuta

        {\Free(A)} && {\Free(C)} \\

        \arrow["{\Free i}", hook, from=1-1, to=1-3]

    \end{tikzcd}\]
\end{theorem}

    ($\implies$) Seja $f \colon A \to B$ um encoding total. Primeiramente, provarenos que $\Free f$
    é uma fibração.

    Seja, então, $p \colon \Free (y_0) \to \Free \left( f (b) \right)$ um caminho em $\Free(B)$.
    Sabemos que $p$ é uma palavra formada pela composição de arestas de $B$ da seguinte forma:

    \begin{equation*}
        y_0 \to y_1 \to y_2 \to \cdots \to y_{n-1} \to f(b)
    \end{equation*}

    Do fato de que $f$ é encoding, tira-se que existe $x_{n-1} \in A$ tal que existe aresta

    em $A$

    \begin{equation*}
        p' \colon x_0 \to x_1 \to \cdots \to x_{n-1} \to b

    identidade. Logo, pela própria construção de $\Free(f)$, $p$ também é identidade; ou seja,
    também é isomorfismo. Logo, o funtor $\Free(f)$ é conservativo.
